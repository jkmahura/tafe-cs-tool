// ════════════════════════════════════════════════════
// TAFE East Zone CS Tool — Google Apps Script
// STEP 1: Replace PASTE_YOUR_SHEET_ID_HERE with your
//         Google Sheet ID (from the Sheet URL)
// STEP 2: Save (Ctrl+S), then Deploy → New version
// ════════════════════════════════════════════════════

var SHEET_ID = 'PASTE_YOUR_SHEET_ID_HERE';

function doPost(e) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var raw = '{}';
    if (e.postData) raw = e.postData.contents || '{}';
    if (e.parameter && e.parameter.payload) raw = e.parameter.payload;
    var data = JSON.parse(raw);
    var action = data.action || 'entry';

    if (action === 'ping' || action === 'test') {
      var pt = ss.getSheetByName('Connection_Test') || ss.insertSheet('Connection_Test');
      pt.getRange(1,1).setValue('Last ping: ' + new Date().toLocaleString());
      return ok();
    }

    // Helper: find row index (1-based) whose EntryId matches
    function findRowById(sheet, oldId) {
      if (!oldId || !sheet || sheet.getLastRow() < 2) return -1;
      var hdrs = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]
                     .map(function(h){ return String(h).toLowerCase().replace(/[^a-z0-9]/g,''); });
      var eidCol = hdrs.indexOf('entryid');
      var finalCol = hdrs.indexOf('isfinal');
      if (eidCol < 0) return -1;
      var vals = sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn()).getValues();
      for (var r = 0; r < vals.length; r++) {
        if (String(vals[r][eidCol]) === String(oldId)) return {row: r + 2, finalCol: finalCol + 1};
      }
      return -1;
    }

    if (action === 'delete') {
      var sheet = ss.getSheetByName('Entries');
      if (sheet && data._replaces) {
        var found = findRowById(sheet, data._replaces);
        if (found !== -1 && found.finalCol > 0)
          sheet.getRange(found.row, found.finalCol).setValue('FALSE');
      }
      return ok();
    }

    if (action === 'entry') {
      var sheet = ss.getSheetByName('Entries') || ss.insertSheet('Entries');
      var HEADERS = ['Date','Month','Year','OutletId','OutletName','OutletType',
        'State','RM','AM','SE','SubmittedBy','EmpId','Role','OnBehalfOf',
        'SL','FS','MJ','Revenue','POil','PParts','SOil','SParts','EW','UC',
        'SMS%','Retention%','VIF','MPReq','MPAvail','Trained','Untrained',
        'Highlight','Remarks','Action','NextVisit','IMOCount',
        'EntryId','ReplacesId','IsFinal','Timestamp'];
      // Always ensure header row matches — fixes column mismatch from old GAS versions
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(HEADERS);
      } else {
        var existingHdrs = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
        var needsUpdate = existingHdrs.length !== HEADERS.length ||
          HEADERS.some(function(h, i) { return String(existingHdrs[i]||'') !== h; });
        if (needsUpdate) {
          // Extend columns if needed, then write correct headers
          if (sheet.getLastColumn() < HEADERS.length)
            sheet.getRange(1, sheet.getLastColumn()+1, 1, HEADERS.length - sheet.getLastColumn()).setValues([HEADERS.slice(sheet.getLastColumn())]);
          sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
        }
      }
      // Mark the old row as not-final (preserve audit trail, don't delete)
      if (data._replaces) {
        var found = findRowById(sheet, data._replaces);
        if (found !== -1 && found.finalCol > 0)
          sheet.getRange(found.row, found.finalCol).setValue('FALSE');
      }
      var k = data.kpi || {}, mp = data.manpower || {};
      sheet.appendRow([data.date, data.month, data.year,
        data.outletId, data.outletName||'', data.outletType,
        data.state, data.rm, data.am, data.se,
        data.by, data.byEid||'', data.byRole, data.onBehalfOf||'',
        k.sl||0, k.fs||0, k.mj||0, k.rev||0, k.poil||0, k.pparts||0,
        k.soil||0, k.sparts||0, k.ew||0, k.uc||0,
        k.sms||0, k.sret||0, k.vif||0,
        mp.req||0, mp.avail||0, mp.trained||0, mp.untrained||0,
        data.highlight||'', data.remarks||'', data.action||'',
        data.nextVisit||'', (data.imo||[]).length,
        data.id||'', data._replaces||'', 'TRUE', new Date().toLocaleString()]);
    }
    return ok();
  } catch(err) {
    return ContentService.createTextOutput(
      JSON.stringify({status:'error',message:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var action = (e.parameter && e.parameter.action) || 'ping';
    var ss = SpreadsheetApp.openById(SHEET_ID);

    if (action === 'pull' || action === 'json') {
      var sheet = ss.getSheetByName('Entries');
      if (!sheet || sheet.getLastRow() < 2) return ok([]);
      var allRows = (e.parameter && e.parameter.all === '1');
      var rows = sheet.getDataRange().getValues();
      var hdrs = rows[0].map(function(h){ return String(h).toLowerCase().replace(/[^a-z0-9]/g,''); });
      var entries = rows.slice(1)
        .filter(function(row) {
          var o = {}; hdrs.forEach(function(h,i){ o[h]=row[i]; });
          // By default only return final (non-superseded) entries
          if (!allRows) {
            var fin = String(o['isfinal']||'').toUpperCase();
            if (fin === 'FALSE') return false;
          }
          return true;
        })
        .map(function(row) {
          var o = {}; hdrs.forEach(function(h,i){ o[h] = row[i]; });
          // Google Sheets returns numeric cells as numbers — coerce to string
          var rawOid = String(o['outletid']||'').trim();
          var rawDate = o['date'];
          // Date cells in Sheets come back as Date objects — format as YYYY-MM-DD
          if (rawDate instanceof Date) {
            var dd = rawDate;
            rawDate = dd.getFullYear()+'-'+String(dd.getMonth()+1).padStart(2,'0')+'-'+String(dd.getDate()).padStart(2,'0');
          } else { rawDate = String(rawDate||'').trim(); }
          return {
            id: String(o['entryid']||'').trim() || (rawOid+'_'+rawDate+'_'+(o['submittedby']||'')),
            date: rawDate, month: String(o['month']||''),
            year: String(o['year']||''), outletId: rawOid,
            outletName: o['outletname']||'', outletType: o['outlettype']||'',
            state: o['state']||'', rm: o['rm']||'', am: o['am']||'', se: o['se']||'',
            by: o['submittedby']||'', byEid: o['empid']||'', byRole: o['role']||'', onBehalfOf: o['onbehalfof']||'',
            isFinal: String(o['isfinal']||'TRUE').toUpperCase() !== 'FALSE',
            kpi: { sl:+o['sl']||0, fs:+o['fs']||0, mj:+o['mj']||0, rev:+o['revenue']||0, poil:+o['poil']||0,
                   pparts:+o['pparts']||0, soil:+o['soil']||0, sparts:+o['sparts']||0,
                   ew:+o['ew']||0, uc:+o['uc']||0, sms:+o['sms%']||+o['sms']||0, sret:+o['retention%']||+o['retention']||0, vif:+o['vif']||0 },
            manpower: { req:+o['mpreq']||0, avail:+o['mpavail']||0,
                        trained:+o['trained']||0, untrained:+o['untrained']||0 },
            highlight: o['highlight']||'', remarks: o['remarks']||'',
            action: o['action']||'', nextVisit: o['nextvisit']||'',
            _replaces: o['replacesid']||''
          };
        });
      return ok(entries);
    }

    // -- DEDUPE: mark duplicate rows IsFinal=FALSE, keep only latest per OutletId+Date+SubmittedBy
    if (action === 'dedupe') {
      var sheet = ss.getSheetByName('Entries');
      if (!sheet || sheet.getLastRow() < 2) return ok({deduped:0});
      var rows = sheet.getDataRange().getValues();
      var hdrs = rows[0].map(function(h){ return String(h).toLowerCase().replace(/[^a-z0-9]/g,''); });
      var outletCol = hdrs.indexOf('outletid');
      var dateCol   = hdrs.indexOf('date');
      var byCol     = hdrs.indexOf('submittedby');
      var empCol    = hdrs.indexOf('empid'); // preferred — immune to name spelling
      var finalCol  = hdrs.indexOf('isfinal');
      var tsCol     = hdrs.indexOf('timestamp');
      if (outletCol<0||dateCol<0||byCol<0||finalCol<0) {
        return ContentService.createTextOutput(JSON.stringify({status:'error',message:'Required column not found. Headers: '+hdrs.join(',')})).setMimeType(ContentService.MimeType.JSON);
      }
      // Find the best (latest timestamp) row per group
      // Group key uses empid when available, falls back to submittedby name
      var best = {};
      for (var i = 1; i < rows.length; i++) {
        var fin = String(rows[i][finalCol]||'').toUpperCase();
        if (fin === 'FALSE') continue;
        var byVal = (empCol>=0 && rows[i][empCol]) ? String(rows[i][empCol]) : String(rows[i][byCol]);
        var key = String(rows[i][outletCol])+'_'+String(rows[i][dateCol])+'_'+byVal;
        var ts = tsCol >= 0 ? new Date(rows[i][tsCol]||0).getTime() : 0;
        if (!best[key] || ts > best[key].ts) best[key] = {row: i+1, ts: ts};
      }
      // Mark all non-best TRUE rows as FALSE
      var deduped = 0;
      for (var i = 1; i < rows.length; i++) {
        var fin = String(rows[i][finalCol]||'').toUpperCase();
        if (fin === 'FALSE') continue;
        var byVal = (empCol>=0 && rows[i][empCol]) ? String(rows[i][empCol]) : String(rows[i][byCol]);
        var key = String(rows[i][outletCol])+'_'+String(rows[i][dateCol])+'_'+byVal;
        if (best[key] && best[key].row !== i+1) {
          sheet.getRange(i+1, finalCol+1).setValue('FALSE');
          deduped++;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({status:'ok',deduped:deduped})).setMimeType(ContentService.MimeType.JSON);
    }

    return ok();
  } catch(err) {
    return ContentService.createTextOutput(
      JSON.stringify({status:'error',message:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function ok(entries) {
  var d = entries !== undefined ? {status:'ok',entries:entries} : {status:'ok'};
  return ContentService.createTextOutput(JSON.stringify(d))
    .setMimeType(ContentService.MimeType.JSON);
}

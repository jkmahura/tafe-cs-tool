☁ Google Sheet Sync

Connect to Google Sheets so every daily submission is automatically saved as a row in your sheet. Follow the steps below to set it up once.

⚠ Note: Google Apps Script blocks response reading from browsers (CORS restriction). After tapping Test, check your Google Sheet directly — if an Entries sheet appeared with a row of data, the connection is working.
📋 Apps Script Setup Guide (do once on laptop)
Step 1 — Create a Google Sheet
Open sheets.google.com → New blank sheet → Name it TAFE East Zone CS – Field Data
Copy your Sheet ID from the URL bar — it is the long string between /d/ and /edit
Step 2 — Open Apps Script
In your Sheet → click Extensions → Apps Script
Delete all existing code in the editor
Step 3 — Paste this code
Tap the Copy button → paste into Apps Script editor → replace YOUR_GOOGLE_SHEET_ID_HERE on line 3 with your actual Sheet ID
// ════════════════════════════════════════════════════
// TAFE East Zone CS Tool — Google Apps Script
// STEP 1: Replace PASTE_YOUR_SHEET_ID_HERE with your
//         Google Sheet ID (from the Sheet URL)
// STEP 2: Save (Ctrl+S), then Deploy → New version
// ════════════════════════════════════════════════════

var SHEET_ID = 'PASTE_YOUR_SHEET_ID_HERE';

function doPost(e) {
  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var raw = '{}';
    if (e.postData) raw = e.postData.contents || '{}';
    if (e.parameter && e.parameter.payload) raw = e.parameter.payload;
    var data = JSON.parse(raw);
    var action = data.action || 'entry';

    if (action === 'ping' || action === 'test') {
      var pt = ss.getSheetByName('Connection_Test') || ss.insertSheet('Connection_Test');
      pt.getRange(1,1).setValue('Last ping: ' + new Date().toLocaleString());
      return ok();
    }

    if (action === 'entry') {
      var sheet = ss.getSheetByName('Entries') || ss.insertSheet('Entries');
      if (sheet.getLastRow() === 0) {
        sheet.appendRow(['Date','Month','Year','OutletId','OutletName','OutletType',
          'State','RM','AM','SE','SubmittedBy','Role','OnBehalfOf',
          'SL','Revenue','POil','PParts','SOil','SParts','EW',
          'SMS%','Retention%','VIF','MPReq','MPAvail','Trained','Untrained',
          'Highlight','Remarks','Action','NextVisit','EntryId','Timestamp']);
      }
      var k = data.kpi || {}, mp = data.manpower || {};
      sheet.appendRow([data.date, data.month, data.year,
        data.outletId, data.outletName||'', data.outletType,
        data.state, data.rm, data.am, data.se,
        data.by, data.byRole, data.onBehalfOf||'',
        k.sl||0, k.rev||0, k.poil||0, k.pparts||0,
        k.soil||0, k.sparts||0, k.ew||0,
        k.sms||0, k.sret||0, k.vif||0,
        mp.req||0, mp.avail||0, mp.trained||0, mp.untrained||0,
        data.highlight||'', data.remarks||'', data.action||'',
        data.nextVisit||'', data.id||'', new Date().toLocaleString()]);
    }
    return ok();
  } catch(err) {
    return ContentService.createTextOutput(
      JSON.stringify({status:'error',message:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var action = (e.parameter && e.parameter.action) || 'ping';
    var ss = SpreadsheetApp.openById(SHEET_ID);

    if (action === 'pull' || action === 'json') {
      var sheet = ss.getSheetByName('Entries');
      if (!sheet || sheet.getLastRow() < 2) return ok([]);
      var rows = sheet.getDataRange().getValues();
      var hdrs = rows[0].map(function(h){ return String(h).toLowerCase().replace(/[^a-z0-9]/g,''); });
      var entries = rows.slice(1).map(function(row) {
        var o = {}; hdrs.forEach(function(h,i){ o[h] = row[i]; });
        return {
          id: o['entryid'] || (o['outletid']+'_'+o['date']+'_'+o['submittedby']),
          date: String(o['date']||''), month: String(o['month']||''),
          year: String(o['year']||''), outletId: o['outletid']||'',
          outletName: o['outletname']||'', outletType: o['outlettype']||'',
          state: o['state']||'', rm: o['rm']||'', am: o['am']||'', se: o['se']||'',
          by: o['submittedby']||'', byRole: o['role']||'', onBehalfOf: o['onbehalfof']||'',
          kpi: { sl:+o['sl']||0, rev:+o['revenue']||0, poil:+o['poil']||0,
                 pparts:+o['pparts']||0, soil:+o['soil']||0, sparts:+o['sparts']||0,
                 ew:+o['ew']||0, sms:+o['sms']||0, sret:+o['retention']||0, vif:+o['vif']||0 },
          manpower: { req:+o['mpreq']||0, avail:+o['mpavail']||0,
                      trained:+o['trained']||0, untrained:+o['untrained']||0 },
          highlight: o['highlight']||'', remarks: o['remarks']||'',
          action: o['action']||'', nextVisit: o['nextvisit']||''
        };
      });
      return ok(entries);
    }
    return ok();
  } catch(err) {
    return ContentService.createTextOutput(
      JSON.stringify({status:'error',message:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function ok(entries) {
  var d = entries !== undefined ? {status:'ok',entries:entries} : {status:'ok'};
  return ContentService.createTextOutput(JSON.stringify(d))
    .setMimeType(ContentService.MimeType.JSON);
}
Step 4 — Save & Deploy
  1. Click 💾 Save (floppy disk icon)
  2. Click Deploy → New Deployment
  3. Click ⚙️ gear icon → select Web App
  4. Set Execute as: Me
  5. Set Who has access: Anyone
  6. Click Deploy
  7. Click Authorize access → choose your Google account → Allow
  8. Copy the Web App URL shown
Step 5 — Paste URL above
Paste the Web App URL in the Google Sheet Sync card above → Save → Test Connection. You should see ✅ Connected.
